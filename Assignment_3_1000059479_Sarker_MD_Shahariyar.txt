                c += 1;
            }

            i += 1;
        }
        if (c == 2) {
            System.out.print(k + " is a prime number\n");
        } else if (c > 2) {
            System.out.print(k + " is not a prime number\n");
        }
        int sum = 0;
        int j = 1;
        while(j <= k){
            if (k % j == 0) {
                sum = sum + j;
            }
            j++;
        }
        int perfect = sum - k;
        if(perfect == k){
            System.out.print(k + " is a perfect number");
