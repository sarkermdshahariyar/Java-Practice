import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Enter a starting number: ");
        int start = scanner.nextInt();
        System.out.println("Enter a ending number: ");
        int end = scanner.nextInt();
        for(int i = start; i<= end; i++ ){
            int count = 0;
            int temp = i;
            int TEMP = i;
            int check = i;
            double sum = 0;
            while(temp >0){
                temp = temp /10;
                count++;
            }
            while(TEMP > 0){
                sum = sum + Math.pow(TEMP%10,count);
                TEMP = TEMP/10;
            }
            if(sum == check){
                System.out.println("Since the sum equals the original number, " + check +" is an Armstrong number.");
            }
        }
    }
}

import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Enter number of students to check: ");
        int k = scanner.nextInt();
        int i = 0;

        while(i < k){
            System.out.println("Enter student ID: ");
            int ID = scanner.nextInt();
            int j = 0;
            boolean lucky = false;
            while(j <= ID){
                if(Math.pow(2,j) == ID){
                    lucky = true;
                    break;
                }
                j++;
            }
            if(lucky){
                System.out.println("Lucky ID: ");
            }
            else{
                System.out.println("Not lucky ID: ");
            }

            i++;
        }
    }
}

import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("How many members: ");
        int mem = scanner.nextInt();
        for(int i = 1; i <=mem; i++){
            System.out.print("Number of exerise for Member " + i + ":" );
            int num_exe = scanner.nextInt();
            if(num_exe > 3){
                System.out.print("Exerise can't be more the 3");
                continue;
            }
            double sum =0 ;
            for(int j =1; j <= num_exe; j++){
                System.out.print("Exercise "+ j + ": ");
                int count = scanner.nextInt();
                if(count > 350){
                    sum = sum + count * 1.5;
                }
                else if(count < 200){
                    sum = sum + count * 0.9;
                }
                else{
                    sum = sum + count;
                }

            }
            double avg = sum /num_exe;
            if(avg > 400){
                avg = avg + 50;
            }
            System.out.print(avg);
            System.out.println();
        }
    }
}
