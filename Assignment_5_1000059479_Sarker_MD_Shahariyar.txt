import java.util.Scanner;
public class A5 {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Num: ");
        int num = scanner.nextInt();
        for(int row = 1; row < 2 * num; row++) {
            if(row > num){
                for(int space= 0; space < num - (2* num - row); space++){
                    System.out.print(" ");
                }
                for(int col = 1; col <= 2* num - row; col++){
                    System.out.print(col);
                }
                for(int col = 2* num - row -1;  col >=1; col-- ){
                    System.out.print(col);
                }
            }
            else{
                for(int space =0; space < num - row; space++){
                    System.out.print(" ");
                }
                for(int col = 1; col <= 2 * row -1; col++){
                    System.out.print(col);
                }
            }
            System.out.println();
        }
    }
}

import java.util.Scanner;
public class A5Q7 {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Num: ");
        int num = scanner.nextInt();
        for (int row = 1; row <= num; row++) {
            int count;
            if (row <= (num / 2) + 1) {
                count = num - 2 * (row - 1);
            } else {
                count = 1 + 2 * (row - ((num / 2) + 1));
            }
            for (int space = 0; space < (num - count) / 2; space++) {
                System.out.print(" ");
            }
            for (int col = 1; col <= count; col++) {
                System.out.print(col);
            }
            System.out.println();
        }
    }
}
